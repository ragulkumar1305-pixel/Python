Sum of Even Digits
input:1234
output:6
program
2.n = int(input())
total = 0

while n > 0:
    digit = n % 10
    if digit % 2 == 0:
        total += digit
    n //= 10

print(total)

Reverse a Number
input:1234
output:4321
program
4.n = int(input())
reverse = 0

while n > 0:
    digit = n % 10
    reverse = reverse * 10 + digit
    n = n // 10

print(reverse)

Palindrome Number
input:121
output:Palindrome
program
5.n = int(input())
original = n
reverse = 0

while n > 0:
    digit = n % 10
    reverse = reverse * 10 + digit
    n = n // 10

if original == reverse:
    print("Palindrome")
else:
    print("Not Palindrome")
